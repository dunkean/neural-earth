"""A4 geography contract: stable fields, physical input units and large masses."""
import unittest
import hashlib

import numpy as np

from terrain_macro import BOOTSTRAP_VERSION, MACRO_VERSION, ContinentalBootstrap, make_macro_factory
from terrain_bootstrap_preview import conditioning_source_files, topology, verify_cpu_sources


class ContinentalBootstrapTests(unittest.TestCase):
    def test_capsule_port_algebra(self):
        # Varying-radius projection and finite rounded ends, from Coast::distance.
        x = np.array([-2., 0., 5., 10., 12.])
        y = np.zeros(5)
        d = ContinentalBootstrap.capsule_distance(x, y, (0, 0, 10, 0, 2, 4))
        np.testing.assert_array_equal(d, [0, 2, 3, 4, 2])

    def test_global_topology_for_distinct_seeds(self):
        # Offset, different-resolution grid from the sea-offset calibration.
        xs = (np.arange(509)+.29)*40e6/509-20e6
        ys = (np.arange(253)+.71)*20e6/253-10e6
        for seed in (0, 42, 20261007):
            fields = make_macro_factory(seed, "A4").sample(xs, ys)
            self.assertTrue(np.isfinite(fields).all())
            metrics = topology(fields[0], 78125)
            self.assertGreater(metrics["land_fraction"], .28)
            self.assertLess(metrics["land_fraction"], .36)
            self.assertGreater(metrics["top5_fraction_of_land"], .95)
            self.assertLessEqual(metrics["land_components"], 20)
            self.assertGreater(metrics["largest_ocean_fraction_of_water"], .90)
            self.assertGreater(metrics["land_1500m_fraction"], .04)
            self.assertLess(metrics["land_0_100m_fraction"], .12)
            factory = make_macro_factory(seed, "A4")
            support = np.asarray(factory.natural.stats["data_quantile_tables"][0])
            self.assertGreaterEqual(fields[0].min(), support[0]-.001)
            self.assertLessEqual(fields[0].max(), support[-1]+.001)
            self.assertFalse(np.any(fields[0] == support[-1]))
            for i, (lo, hi) in enumerate(factory.climate_support):
                self.assertGreaterEqual(fields[i+1].min(), lo-.001)
                self.assertLessEqual(fields[i+1].max(), hi+.001)

    def test_shore_gradient_at_coarse_cell_centres(self):
        distances = np.array([-15360, -7680, -3840, 0, 3840, 7680, 15360.])
        for height in (65., 300., 1000., 5500.):
            for broad in (-.4, 0, .4):
                for detail in (-.4, .4):
                    elev = ContinentalBootstrap.shore_elevation(distances, height, broad, detail)
                    self.assertEqual(elev[3], 0)
                    self.assertLess(elev[2], -50)
                    self.assertLess(elev[1], -100)
                    self.assertLess(elev[0], -200)
                    self.assertGreater(elev[4], 35)
                    self.assertGreater(elev[5], 55)
                    self.assertTrue(np.all(np.diff(elev) >= 0))

    def test_soft_cap_retains_crest_variation(self):
        values = np.linspace(50, 10000, 1000)
        capped = ContinentalBootstrap.soft_land_cap(values, 5814.)
        self.assertTrue(np.all(np.diff(capped) > 0))
        self.assertLess(capped.max(), 5814.)
        np.testing.assert_array_equal(capped[values <= .7*5814], values[values <= .7*5814])

    def test_cpu_guard_includes_stats_world_and_visual_sources(self):
        paths = conditioning_source_files()
        for name in ("terrain_world.py", "synthetic_map_stats.json", "terrain_bootstrap_preview.py"):
            self.assertIn(name, paths)
        report = {"source_sha256": {name: hashlib.sha256(path.read_bytes()).hexdigest()
                                   for name, path in paths.items()}}
        verify_cpu_sources(report)
        for name in ("terrain_world.py", "synthetic_map_stats.json", "terrain_bootstrap_preview.py"):
            altered = {"source_sha256": dict(report["source_sha256"])}
            altered["source_sha256"][name] = "0"*64
            with self.assertRaisesRegex(ValueError, name):
                verify_cpu_sources(altered)

    def test_crop_invariant_physical_and_model_input(self):
        factory = make_macro_factory(42, "A4")
        whole = factory.sample_raw(-150, -270, -60, -180)
        crop = factory.sample_raw(-125, -240, -95, -210)
        np.testing.assert_array_equal(whole[:, 30:60, 25:55], crop)
        model = factory(-125, -240, -95, -210).numpy()
        np.testing.assert_array_equal(model[1:], crop[1:])
        np.testing.assert_array_equal(model[0], np.sign(crop[0])*np.sqrt(np.abs(crop[0])))

    def test_distinct_manifest_versions(self):
        from terrain_manifest import build_manifest
        reference = build_manifest(0, "A0", file_hashes=False)
        old = build_manifest(0, "A3", file_hashes=False)
        candidate = build_manifest(0, "A4", file_hashes=False)
        self.assertEqual(old["geography"]["macro_version"], MACRO_VERSION)
        self.assertEqual(candidate["geography"]["macro_version"], BOOTSTRAP_VERSION)
        self.assertEqual(candidate["conditioning"]["cond_snr"], [.5]*5)
        self.assertTrue(candidate["conditioning"]["regional_procedures"])
        self.assertEqual(reference["world_profile"], "natural")
        self.assertEqual(len({m["world_hash"] for m in (reference, old, candidate)}), 3)


if __name__ == "__main__":
    unittest.main()
