"""CPU comparison of attested historical baselines and the new A4 NN run."""
import argparse
import json
from pathlib import Path

import numpy as np
from PIL import Image

from terrain_bootstrap_preview import panel
from terrain_bootstrap_shore import _attest_artifact, _sha


def compare(old_report, new_report):
    old_report, new_report = Path(old_report).resolve(), Path(new_report).resolve()
    old, new = (json.loads(p.read_text(encoding="utf-8")) for p in (old_report, new_report))
    output = new_report.parent
    receipt = {"stage": "cpu-artifact-comparison-only", "gpu_rerun": False,
               "historical_report_sha256": _sha(old_report), "current_report_sha256": _sha(new_report),
               "exporter_sha256": _sha(__file__), "sites": {}}
    for name, current in new["sites"].items():
        if name not in old["sites"]:
            continue
        baseline = old["sites"][name]
        if baseline["site"] != current["site"]:
            raise ValueError("Comparison sites differ")
        board = Image.new("RGB", (1024*4, 1054), (18, 24, 30))
        products = [("A0 historical V1 run", "A0", old_report.parent, baseline),
                    ("A3 historical V1 run", "A3", old_report.parent, baseline),
                    ("A4 V1 historical", "A4", old_report.parent, baseline),
                    ("A4 V2 new NN", "A4", new_report.parent, current)]
        row = {}
        for col, (label, profile, root, record) in enumerate(products):
            artifact = record["profiles"][profile]["artifact"]
            attestation = _attest_artifact(root, artifact, record["site"], profile)
            with np.load(root/artifact["npz"], allow_pickle=False) as data:
                elev, bounds = data["dem_m"], data["native_bounds"]
            if col == 0:
                expected = bounds
            elif not np.array_equal(expected, bounds):
                raise ValueError("DEM footprints differ")
            fields = np.broadcast_to(elev, (5, *elev.shape))
            board.paste(panel(fields, 30, f"{label} {name} | {attestation['manifest_hash'][:12]}"), (col*1024, 0))
            row[label] = {"attestation": attestation, "artifact_root": str(root),
                          "near_sea_fraction_minus1_plus1m": float(np.mean(np.abs(elev) < 1)),
                          "opus_v1_band_fraction_minus0_09_plus0_82m": float(np.mean((elev > -.09) & (elev < .82)))}
        target = output/f"{name}-historical-v2-comparison.png"
        board.save(target)
        row["comparison_png_sha256"] = _sha(target)
        receipt["sites"][name] = row
    (output/"historical-comparison.json").write_text(json.dumps(receipt, indent=2), encoding="utf-8")
    return receipt


def compare_arrays(previous_report, final_report):
    """Attest raw stage equality after metadata/source-only changes."""
    previous_report, final_report = Path(previous_report).resolve(), Path(final_report).resolve()
    old, new = (json.loads(p.read_text(encoding="utf-8")) for p in (previous_report, final_report))
    result = {"stage": "cpu-comparison-only", "gpu_rerun": False,
              "previous_report_sha256": _sha(previous_report), "final_report_sha256": _sha(final_report),
              "comparer_sha256": _sha(__file__), "sites": {}}
    names = ("dem_m", "climate", "coarse_weighted", "coarse_unweighted",
             "latent_weighted", "latent_normalized", "latent_lowfreq")
    for site_id, previous in old["sites"].items():
        current = new["sites"][site_id]
        if previous["site"] != current["site"]:
            raise ValueError("Original-site coordinates changed")
        old_artifact, new_artifact = (r["profiles"]["A4"]["artifact"] for r in (previous, current))
        old_attested = _attest_artifact(previous_report.parent, old_artifact, previous["site"], "A4")
        new_attested = _attest_artifact(final_report.parent, new_artifact, current["site"], "A4")
        arrays = {}
        with np.load(previous_report.parent/old_artifact["npz"], allow_pickle=False) as a, np.load(final_report.parent/new_artifact["npz"], allow_pickle=False) as b:
            for name in names:
                if a[name].shape != b[name].shape:
                    raise ValueError("Compared footprints differ")
                arrays[name] = {"dtype_previous": str(a[name].dtype), "dtype_final": str(b[name].dtype),
                    "shape": list(a[name].shape), "byte_identical_raw_array": a[name].tobytes() == b[name].tobytes(),
                    "max_absolute_difference": float(np.max(np.abs(a[name].astype(np.float64)-b[name].astype(np.float64))))}
        result["sites"][site_id] = {"previous_attestation": old_attested, "final_attestation": new_attested,
                                    "arrays": arrays}
    target = final_report.parent/"source-only-array-comparison.json"
    target.write_text(json.dumps(result, indent=2), encoding="utf-8")
    return result


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--historical-report", type=Path, required=True)
    parser.add_argument("--current-report", type=Path, required=True)
    parser.add_argument("--compare-raw-arrays", action="store_true")
    args = parser.parse_args()
    (compare_arrays if args.compare_raw_arrays else compare)(args.historical_report, args.current_report)
